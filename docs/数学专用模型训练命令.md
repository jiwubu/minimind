
# 前提：dataset/pretrain_t2t_mini.jsonl(1.2GB) 与 sft_t2t_mini.jsonl(1.7GB)
# 不入 git，需提前同步到本机；算术数据在下面现生成

```bash
# =========================
# 环境准备
# =========================
ln -sf /usr/bin/python3 /usr/bin/python
ln -sf /usr/bin/pip3 /usr/bin/pip
yum install tmux

python -m venv .venv
source .venv/bin/activate

pip install --upgrade pip
pip install -r requirements.txt

# =========================
# 数据生成（§3；dataset/*.jsonl 不入 git，服务器上现生成，约几分钟）
# =========================
python math/gen_math_data_addsub.py \
  --out dataset/sft_t2t_math_addsub_v3.jsonl \
  --n 400000 --max_digits 8 --neg_ratio 0.25 \
  --mix_general dataset/sft_t2t_mini.jsonl --mix_ratio 0.3

python math/gen_math_data_muldiv.py \
  --out dataset/sft_t2t_math_muldiv_v3.jsonl \
  --n 400000 \
  --mix_general dataset/sft_t2t_mini.jsonl --mix_ratio 0.3

cat dataset/sft_t2t_math_addsub_v3.jsonl dataset/sft_t2t_math_muldiv_v3.jsonl > dataset/sft_t2t_math_all_v3.jsonl
wc -l dataset/sft_t2t_math_all_v3.jsonl    # 应为 1,142,856

# =========================
# 预训练（§2.2，从零，64M 默认配置）
# 仅全新环境需要；已有 out/pretrain_math_768.pth 时跳过本段直接进 SFT
#（重跑会覆盖已有预训练权重并浪费数小时）
# =========================
cd trainer

#多卡训练的完整命令见下方「多卡训练」段（torchrun 启动，训练参数与单卡相同）
python train_pretrain.py \
  --data_path ../dataset/pretrain_t2t_mini.jsonl \
  --from_weight none \
  --save_weight pretrain_math \
  --max_seq_len 340 \
  --epochs 2 \
  --batch_size 32 \
  --dtype bfloat16 \
  --from_resume 1

# =========================
# 合并 SFT（§7.3：四则一起训；max_seq_len=1344 由乘除轨迹长度决定，
# 576 会把乘除样本全部截断）
# from_resume 1 仅在数据与本权重已训部分完全相同（同 seed 同参数）时用于续训；
# 若换了数据或轨迹格式（如 math_format v2），必须 --from_resume 0 并换新权重名
# 从 pretrain_math 从头训，否则新旧格式会混串
# =========================
python train_full_sft.py \
  --data_path ../dataset/sft_t2t_math_all_v3.jsonl \
  --from_weight pretrain_math \
  --save_weight full_sft_math_all_v3 \
  --epochs 1 \
  --batch_size 32 \
  --learning_rate 2e-5 \
  --max_seq_len 1344 \
  --dtype bfloat16 \
  --num_workers 8 \
  --from_resume 1

# =========================
# 多卡训练（单机 2/4 卡，可选）
# =========================
# torchrun 启动，每卡一个进程：设备按 LOCAL_RANK 自动映射（无需手动指定
# --device），数据由 DistributedSampler 自动分片。
# 全局有效 batch = batch_size × 卡数（2 卡 × 32 = 64）；学习率保持单卡值即可，
# 若要严格等效单卡可将 batch_size 减半。--from_resume 续训在多卡下同样适用。
# 若 29500 端口被占，给 torchrun 加 --master_port=29512。

# 预训练（2 卡；4 卡把 --nproc_per_node 改为 4）
cd trainer
torchrun --nproc_per_node 2 train_pretrain.py \
  --data_path ../dataset/pretrain_t2t_mini.jsonl \
  --from_weight none \
  --save_weight pretrain_math \
  --max_seq_len 340 \
  --epochs 2 \
  --batch_size 32 \
  --dtype bfloat16 \
  --from_resume 1

# 合并 SFT（2 卡）
torchrun --nproc_per_node 2 train_full_sft.py \
  --data_path ../dataset/sft_t2t_math_all_v3.jsonl \
  --from_weight pretrain_math \
  --save_weight full_sft_math_all_v3 \
  --epochs 1 \
  --batch_size 32 \
  --learning_rate 2e-5 \
  --max_seq_len 1344 \
  --dtype bfloat16 \
  --num_workers 8 \
  --from_resume 1

# =========================
# 评测（§7.4；回到仓库根目录执行）
# =========================
cd ..

python math/eval_math.py --weight full_sft_math_all_v3 --max_digits 8 --n 25
#   乘除默认只测训练范围 4 位（摘要不被范围外 0% 拉低）；测外推显式加
#   --mul_digits 6 --div_digits 8。需要本地 eval_math.py 为最新版
#   （含训练范围默认值）。快速迭代用：
# python math/eval_math_fast.py --weight full_sft_math_all_v3   # 约 1/10 耗时

# 方式1：基于 PyTorch 模型（确保./out目录下有对应权重）
python eval_llm.py --weight full_sft_math_all_v3

# 方式2：使用Transformers 格式模型
#转换模型为transformers格式，改 scripts/convert_model.py
torch_path = f"../out/full_sft_math_all_v3_768.pth"          # 原: full_sft_768.pth
transformers_path = "../minimind-math-v3"                     # 原: ../minimind-3

#在 scripts/ 目录下运行
cd /Users/jiwubu/workspace/minimind/scripts
python convert_model.py

python eval_llm.py --load_from ./scripts/minimind-math-v3 

```
